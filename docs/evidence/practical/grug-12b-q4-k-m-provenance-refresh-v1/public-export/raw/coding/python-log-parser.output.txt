import sys
import os

def parse_log(file_path):
    severities = ["ERROR", "WARNING", "CRITICAL"]
    counts = {severity: 0 for severity in severities}
    examples = {severity: [] for severity in severities}

    if not os.path.exists(file_path):
        print(f"Error: File '{file_path}' not found.")
        return

    try:
        with open(file_path, 'r', encoding='utf-8') as f:
            for line in f:
                line = line.strip()
                if not line:
                    continue
                
                for severity in severities:
                    if severity in line:
                        counts[severity] += 1
                        if len(examples[severity]) < 5:
                            examples[severity].append(line)
                        break  # Only count the first match found in line

        print(f"--- Log Summary for {file_path} ---")
        for severity in severities:
            print(f"\n[{severity}] Total: {counts[severity]}")
            if examples[severity]:
                print("First 5 examples:")
                for ex in examples[severity]:
                    print(f"  > {ex}")
            else:
                print("  No matching lines found.")

    except Exception as e:
        print(f"An error occurred while reading the file: {e}")

if __name__ == "__main__":
    if len(sys.argv) != 2:
        print("Usage: python log_parser.py <path_to_log_file>")
    else:
        parse_log(sys.argv[1])
